<script>
// 色定義：上から順に 赤, 黄, 青, オレンジ, 灰色, 灰色
const ROW_COLORS = [
  '#f44336', // 赤
  '#ffeb3b', // 黄
  '#2196f3', // 青
  '#ff9800', // オレンジ
  '#9e9e9e', // 灰色1
  '#616161'  // 灰色2
];

// 譜面データ構造 (1小節内の相対位置 relativeTime で音符のタイミングを保持)
let notes = []; // { bar: 0~127, relativeTime: 0.0~1.0, row: 0~5 }
let gridMode = 16; // 16 or 24
let bpm = 120;
let offsetSeconds = 0;
let undoStack = [];
let redoStack = [];

// Web Audio API Setup
const AudioContext = window.AudioContext || window.webkitAudioContext;
const audioCtx = new AudioContext();

// バスドラム低音再生 (遅延ゼロのオシレーター合成音)
function playKickSound() {
  const seVol = document.getElementById('se-vol').value / 100;
  if (seVol <= 0) return;

  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  
  osc.type = 'sine';
  osc.frequency.setValueAtTime(150, audioCtx.currentTime);
  osc.frequency.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.12);
  
  gain.gain.setValueAtTime(seVol, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.12);
  
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  
  osc.start();
  osc.stop(audioCtx.currentTime + 0.12);
}

// 履歴保存 (Undo/Redo)
function saveHistory() {
  undoStack.push(JSON.stringify(notes));
  if (undoStack.length > 30) undoStack.shift();
  redoStack = [];
}

function undo() {
  if (undoStack.length === 0) return;
  redoStack.push(JSON.stringify(notes));
  notes = JSON.parse(undoStack.pop());
  renderAllLanes();
}

function redo() {
  if (redoStack.length === 0) return;
  undoStack.push(JSON.stringify(notes));
  notes = JSON.parse(redoStack.pop());
  renderAllLanes();
}

function toggleGridMode() {
  gridMode = (gridMode === 16) ? 24 : 16;
  document.getElementById('btn-grid').innerText = gridMode + '分グリッド';
  renderAllLanes();
}

// レーン初期化 (16本)
const container = document.getElementById('editor-container');
const laneCanvases = [];

function initLanes() {
  container.innerHTML = '';
  for (let l = 0; l < 16; l++) {
    const rowDiv = document.createElement('div');
    rowDiv.className = 'lane-row';

    const btn = document.createElement('button');
    btn.className = 'play-here-btn';
    btn.innerText = 'ここから\n再生';
    btn.onclick = () => togglePlayLane(l, btn);

    const canvas = document.createElement('canvas');
    canvas.width = 1200;
    canvas.height = 120;
    
    // タップ・クリックイベント
    canvas.addEventListener('pointerdown', (e) => handleCanvasClick(e, l, canvas));

    rowDiv.appendChild(btn);
    rowDiv.appendChild(canvas);
    container.appendChild(rowDiv);
    laneCanvases.push(canvas);
  }
  renderAllLanes();
}

// Canvas描画ロジック
function drawLane(laneIndex) {
  const canvas = laneCanvases[laneIndex];
  const ctx = canvas.getContext('2d');
  const w = canvas.width;
  const h = canvas.height;
  
  ctx.clearRect(0, 0, w, h);
  
  const barWidth = w / 8; // 8小節
  const rowHeight = h / 6; // 6行

  // 方眼紙・グリッド描画
  ctx.lineWidth = 1;
  for (let b = 0; b < 8; b++) {
    const startX = b * barWidth;
    for (let i = 0; i < gridMode; i++) {
      const x = startX + (i / gridMode) * barWidth;
      ctx.strokeStyle = (i === 0) ? '#666' : '#2a2a2a';
      ctx.beginPath();
      ctx.moveTo(x, 0); ctx.lineTo(x, h);
      ctx.stroke();
    }
  }

  // 横線描画 (2列ごとに太線)
  for (let r = 0; r <= 6; r++) {
    const y = r * rowHeight;
    ctx.lineWidth = (r % 2 === 0) ? 2 : 1;
    ctx.strokeStyle = (r % 2 === 0) ? '#888' : '#333';
    ctx.beginPath();
    ctx.moveTo(0, y); ctx.lineTo(w, y);
    ctx.stroke();
  }

  // ノーツ描画 (不透明度 60% = 0.6)
  const startBar = laneIndex * 8;
  const endBar = startBar + 8;
  const laneNotes = notes.filter(n => n.bar >= startBar && n.bar < endBar);

  ctx.globalAlpha = 0.6; // 仕様: 不透明度60%
  for (let n of laneNotes) {
    const localBar = n.bar - startBar;
    const x = (localBar + n.relativeTime) * barWidth;
    const y = n.row * rowHeight;

    ctx.fillStyle = ROW_COLORS[n.row];
    // 正方形ノーツ
    ctx.fillRect(x + 1, y + 1, rowHeight - 2, rowHeight - 2);
  }
  ctx.globalAlpha = 1.0;
}

function renderAllLanes() {
  for (let i = 0; i < 16; i++) drawLane(i);
}

// ノーツ打込・消去
function handleCanvasClick(e, laneIndex, canvas) {
  if (audioCtx.state === 'suspended') audioCtx.resume();
  
  const rect = canvas.getBoundingClientRect();
  const clickX = (e.clientX - rect.left) * (canvas.width / rect.width);
  const clickY = (e.clientY - rect.top) * (canvas.height / rect.height);

  const barWidth = canvas.width / 8;
  const rowHeight = canvas.height / 6;

  const barOffset = Math.floor(clickX / barWidth);
  const targetBar = laneIndex * 8 + barOffset;
  const relativeX = (clickX % barWidth) / barWidth;

  // 現在のグリッドモードに合わせてスナップ
  const snappedTime = Math.round(relativeX * gridMode) / gridMode;
  const targetRow = Math.floor(clickY / rowHeight);

  if (targetRow < 0 || targetRow >= 6) return;

  saveHistory();

  // 既存ノーツの検索（トグル動作）
  const existingIndex = notes.findIndex(n => 
    n.bar === targetBar && 
    Math.abs(n.relativeTime - snappedTime) < 0.001 && 
    n.row === targetRow
  );

  if (existingIndex >= 0) {
    notes.splice(existingIndex, 1); // 消去
  } else {
    notes.push({ bar: targetBar, relativeTime: snappedTime, row: targetRow }); // 打ち込み
    playKickSound(); // 音を鳴らす
  }

  drawLane(laneIndex);
}

function togglePlayLane(laneIndex, btn) {
  const isPlaying = btn.classList.contains('playing');
  document.querySelectorAll('.play-here-btn').forEach(b => {
    b.classList.remove('playing');
    b.innerText = 'ここから\n再生';
  });

  if (!isPlaying) {
    btn.classList.add('playing');
    btn.innerText = '停止';
  }
}

function togglePlayAll() {
  const btn = document.getElementById('btn-play-all');
  if (btn.innerText === '最初から再生') {
    btn.innerText = '停止';
    btn.className = 'btn-large btn-danger';
  } else {
    btn.innerText = '最初から再生';
    btn.className = 'btn-large btn-primary';
  }
}

// 初期化実行
initLanes();
</script>
